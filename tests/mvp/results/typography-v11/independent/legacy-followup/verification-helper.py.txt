from pathlib import Path

  summary=[]
  for line in text.splitlines():
   if line.startswith('{'):
    try:summary.append(json.loads(line))
    except ValueError:pass
  if name=='ui':
   (out/'ui-components.json').write_bytes(prior.read_bytes());summary.append(json.loads(prior.read_bytes()))
  report['suites'].append({'name':name,'command':cmd,'exit_code':p.returncode,'summary':summary,'diagnostics':[s for s in text.splitlines() if s.startswith(('ERROR:','SCRIPT ERROR:','FAIL:'))],'log_sha256':sha(raw)})
finally:
 prior.write_bytes(saved);report['previous_ui_bytes_preserved']=prior.read_bytes()==saved;report['source_after']=hashes();report['changed_sources']=[p for p,h in report['source_before'].items() if report['source_after'][p]!=h]
