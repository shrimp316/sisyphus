extends SceneTree
const Main = preload("res://scripts/main.gd")
const OUT = "res://tests/mvp/results/typography-v11/independent/legacy-followup/native"
var records: Array = []
func _initialize(): call_deferred("run")
func run():
	if DisplayServer.get_name() == "headless":
		quit(2)
		return
	DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path(OUT))
	for dimensions: Vector2i in [Vector2i(1920,1080),Vector2i(1280,720)]:
		var viewport := SubViewport.new()
		viewport.size = dimensions
		viewport.render_target_update_mode = SubViewport.UPDATE_ALWAYS
		root.add_child(viewport)
		var scene = Main.new()
		scene.save_path = "user://legacy-render-%s.json" % Time.get_ticks_usec()
		scene.test_seed = 101
		scene.test_weather = "clear"
		viewport.add_child(scene)
		scene.set_process(false)
		scene.set_physics_process(false)
		for state: String in ["title","pause","upgrade","result","summit"]:
			scene.screen = "title" if state == "title" else "game"
			scene.paused = state == "pause"
			scene.upgrade_open = state == "upgrade"
			scene.boulder.state = state if state in ["result","summit"] else "climbing"
			scene.summit_quiet = 4.0
			scene.queue_redraw()
			for frame in range(3): await process_frame
			await RenderingServer.frame_post_draw
			var pixels = viewport.get_texture().get_image()
			var file_path = OUT.path_join("%s-%d.png" % [state,dimensions.x])
			var error = pixels.save_png(file_path)
			records.append({"state":state,"viewport":[dimensions.x,dimensions.y],"path":file_path,"error":error,"sha256":FileAccess.get_sha256(file_path)})
		viewport.queue_free()
		await process_frame
	var report = {"platform":OS.get_name(),"display":DisplayServer.get_name(),"renderer":RenderingServer.get_current_rendering_method(),"records":records,"scope":"Actual legacy production Canvas rendering with frozen artificial screen states, seed101 clear; real Windows SubViewport, not OS fullscreen; isolated missing save path; no writes."}
	var file = FileAccess.open(OUT.path_join("render.json"),FileAccess.WRITE)
	file.store_string(JSON.stringify(report,"\t"))
	print(JSON.stringify(report))
	quit(0)
